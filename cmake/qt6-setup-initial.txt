### Código de Ejemplo - main.cpp con Qt6/QML

```cpp
// apps/main.cpp - Reemplaza src/amule-gui.cpp antiguo
#include <QApplication>
#include <QQuickView>
#include "qml/main.qml"
#include <QDebug>

int main(int argc, char *argv[]) {
    QApplication app(argc, argv);
    
    // Animaciones fluidas configuradas desde Qt6 native
    qApp->setNativeEvent((fnptr*)nullptr);  // Para QML aceleración
    
    // Inicializar con transiciones CSS nativas
    QQuickView view(argc, argv);
    view.setSource(QUrl(QString("qrc:/qml/main.qml")));
    
    qDebug() << "Iniciando aMule Qt6 con fluid UI";
    
    return app.exec();
}
```

### Animaciones Fluidas en QML

```qml
// qml/AnimationEngine.qml - Motor de animaciones fluido
import QtQuick 2.15
import QtQuick.Window 2.15
import QtQuick.Controls 2.15

QtPropertyModel {
    // Configuración global para transiciones CSS fluidas
    
    property int "FadeIn": 300;           // Opacidad 0->1 en ms
    property int "SlideUp": 200;          // Movimiento vertical suave
    property int "ScaleSoft": 500;        // Escala sin discontinuidades
    
    // Ejemplo de QML con animaciones CSS nativas
    Rectangle {
        width: 400
        height: 300
        
        PropertyAnimation {
            target: parent
            property: "opacity"
            duration: FadeIn
            from: 0.0
            to: 1.0
            easing.type: Easing.OutQuad  // Curva suave (no lineal)
        }
        
        PointAnimation {
            target: parent
            property: "y"
            duration: SlideUp
            from: 100
            to: 0
            easing.type: Easing.OutBack  // Sin rebote
        }
    }
}
```

### Estilos CSS Modernos en Qt6/QML

```qml
// qml/styles/ModernPalette.qml - Paleta de colores moderna
import QtQuick
import QtQuick.Window

QtPropertyModel {
    // Branding aMule moderno
    property string "primary": "#1c95e0";     // Azul vibrante (#22b9e8)
    property string "secondary": "#43b16d";   // Verde brillante (#57f392)
    property string "accent": "#ffab00";       // Naranja (#ffb86c)
    property string "background": "#f5f5f5";   // Gris claro moderno
    property string "surface": "#ffffff";      // Blanco puro
    
    // Tipografía fluida con Poppins/Roboto
    property string "fontHeadline": "Poppins, sans-serif";
    property string "fontBody": "Roboto, sans-serif";
}

// Uso del paletizado en UI components
Rectangle {
    color: ModernPalette.primary;      // Azul moderno
    width: 400
    height: 300
    
    // Animaciones CSS con easing curves nativas
    PropertyAnimation {
        target: parent
        property: "opacity"
        duration: 300
        from: 0.0
        to: 1.0
        easing.type: Easing.OutQuad  // Suave, no lineal
    }
    
    PointAnimation {
        target: parent
        property: "y"
        duration: 200
        from: 50
        to: 0
        easing.type: Easing.OutBack  // Sin discontinuidades
    }
}
```

### Beneficios Clave vs wxWidgets

| Métrica | wxWidgets (actual) | Qt6/QML (objetivo) |
|---------|-------------------|--------------------|
| Animaciones nativas | Limitadas a QSS básico | CSS completo + hardware accel. |
| Rendering | OpenGL/OpenGL ES | QML declarativo nativo |
| Framerate control | wxApp event loop | `qApp->setNativeEvent` directo |
| Theming | Color hardcodeado | CSS variables + theming |
